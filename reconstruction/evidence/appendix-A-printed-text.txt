T_tent = 90 [F]

T_sky = 55 [F]

rho_ice = density(Ice_inc, T=T_ice)
m_ice = rho_ice * A * thickness
t = 24 [hr]
g = 32.2 [ft/s^2] * convert(ft/s^2, ft/hr^2)
P_atm = 14.7 [psia]
B = 1 / converttemp(F, R, T_film)
B2 = 1 / converttemp(F, R, T_film)
nu = kinematicviscosity(Air, T=T_film, P=P_atm)
nu2 = kinematicviscosity(Air, T=T_film2, P=P_atm)

alpha = thermaldiffusivity(Air, T=T_film, P=P_atm)
alpha2 = thermaldiffusivity(Air, T=T_film2, P=P_atm)
k = conductivity(Air, T=T_film)

cp_water = cp(Water, T=T_final, P=P_atm)
Gs = 291.6 [Btu/hr*ft^2]
sigma = 5.670374419e-8 [W/m^2*K^4]  "Stefan-Boltzmann constant converted to appropriate
units if necessary"
epsilon_tent = 0.87  "Assuming this for the tent material"
epsilon_ice = 0.95   "Assume emissivity for ice"
epsilon = 0.87
L_c = A / (2 * (L + W))
R_a_L_c = g * B * (T_air - T_ice) * (L_c^3) * (1 / (nu * alpha))
Nu_L_c = 0.52 * R_a_L_c^(1/5)
h = (Nu_L_c * k) / L_c
R_a_L_c_2 = g * B2 * (T_tent - T_air) * (L_c^3) * (1 / (nu2 * alpha2))
Nu_L_c_2 = 0.15 * R_a_L_c_2^(1/3)
h_tent = (Nu_L_c_2 * k2) / L_c
"------------------------------------------------------------------------------------------------------------------------------
----------"
"1a) Power required to freeze the ice in Btu/hr for both design conditions (8 hour startup and 24
hour startup)"
"------------------------------------------------------------------------------------------------------------------------------
----------"
HF = enthalpy_fusion(Water)
q_freeze = (m_ice * HF) / t
"------------------------------------------------------------------------------------------------------------------------------
----------"
"1b) Power Required to cool ice to 24F in Btu/hr"
"------------------------------------------------------------------------------------------------------------------------------
----------"
cp_ice = specheat(Ice, T=T_ice, P=P_atm)
q_ice_cool = (m_ice * cp_ice * (T_final_R - T_ice_R)) / t
"------------------------------------------------------------------------------------------------------------------------------
----------"
"1c) Power Required to cool water from 52F to 32F"
"------------------------------------------------------------------------------------------------------------------------------
----------"
q_cool_water = (m_ice * cp_water * (T_initial_R - T_final_R)) / t
"------------------------------------------------------------------------------------------------------------------------------
----------"
"1d) Heat Gain by natural convection on ice surface in Btu/hr"
"------------------------------------------------------------------------------------------------------------------------------
----------"
